cmake_minimum_required(VERSION 3.16)
project(Nitemare3D C)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

if(EMSCRIPTEN)
  # Browser build (see README, "Browser (WebAssembly)"): SDL2 comes from Emscripten's port.
  option(N3D_WASM_THREADS "Use Web Workers for the hi-res renderer (needs a cross-origin isolated page)" OFF)
  set(N3D_WASM_DATA "" CACHE PATH "Game data folder to bundle into the build (private builds only; default: the page asks for the files)")
else()
  find_package(SDL2 QUIET)
  if(NOT SDL2_FOUND AND WIN32)
    # Native Windows build without vcpkg: use SDL's official development package.
    include(FetchContent)
    if(POLICY CMP0135)
      cmake_policy(SET CMP0135 NEW)
    endif()
    if(MSVC)
      FetchContent_Declare(sdl2_devel
        URL https://github.com/libsdl-org/SDL/releases/download/release-2.32.10/SDL2-devel-2.32.10-VC.zip
        URL_HASH SHA256=af347939395a58b365846aaea27391e69f9ec9d4dd650d6ac40802159b418a6e)
      FetchContent_MakeAvailable(sdl2_devel)
      set(SDL2_DIR ${sdl2_devel_SOURCE_DIR}/cmake)
    else()
      FetchContent_Declare(sdl2_devel
        URL https://github.com/libsdl-org/SDL/releases/download/release-2.32.10/SDL2-devel-2.32.10-mingw.zip
        URL_HASH SHA256=f15cff5fca62ec9381a016ef1d42a95c638cd72d2f226ba5781c76fe43dbd1ac)
      FetchContent_MakeAvailable(sdl2_devel)
      set(SDL2_DIR ${sdl2_devel_SOURCE_DIR}/x86_64-w64-mingw32/lib/cmake/SDL2)
    endif()
    find_package(SDL2 REQUIRED CONFIG)
  elseif(NOT SDL2_FOUND)
    find_package(SDL2 REQUIRED)
  endif()
  find_package(Threads)
endif()

if(WIN32)
  option(N3D_D3D12 "Direct3D 12 video backend (default on Windows; --renderer picks SDL's instead)" ON)
endif()

# Recompiled game code generated by tools/recomp.py (see README).
file(GLOB GEN_SOURCES ${CMAKE_SOURCE_DIR}/src/gen/rc_*.c)
if(NOT GEN_SOURCES)
  message(FATAL_ERROR "src/gen is empty: run tools/build_gen.sh first (needs Ghidra and your NITE3W.EXE)")
endif()

add_executable(nitemare3d
  src/host_sdl.c
  src/video_sdl.c
  src/music.c
  src/dls.c
  src/msadlib.c
  third_party/stb_vorbis/stb_vorbis.c
  third_party/nuked_opl3/opl3.c
  src/rt/rt_core.c
  src/rt/dos.c
  src/rt/win16.c
  src/rt/overrides.c
  src/hd.c
  src/hd_tex.c
  src/hd_pack.c
  src/net.c
  src/hud.c
  src/light.c
  ${GEN_SOURCES})

# Gothic HUD art (assets/hud) -> hud.pak next to the program (see README, "Gothic HUD").
find_package(Python3 COMPONENTS Interpreter)
file(GLOB HUD_ASSETS ${CMAKE_SOURCE_DIR}/assets/hud/*.png ${CMAKE_SOURCE_DIR}/assets/hud/*.fnt)
if(Python3_Interpreter_FOUND AND HUD_ASSETS)
  add_custom_command(OUTPUT ${CMAKE_BINARY_DIR}/hud.pak
    COMMAND ${Python3_EXECUTABLE} -I ${CMAKE_SOURCE_DIR}/tools/hud_pack.py ${CMAKE_SOURCE_DIR}/assets/hud ${CMAKE_BINARY_DIR}/hud.pak
    DEPENDS ${HUD_ASSETS} ${CMAKE_SOURCE_DIR}/tools/hud_pack.py
    COMMENT "Packing HUD art")
  add_custom_target(hud_pak ALL DEPENDS ${CMAKE_BINARY_DIR}/hud.pak)
  add_dependencies(nitemare3d hud_pak)
  set(N3D_HUD_PAK ${CMAKE_BINARY_DIR}/hud.pak)
  if(CMAKE_CONFIGURATION_TYPES)
    # Visual Studio puts the program in build/Release; the game looks for hud.pak next to it.
    add_custom_command(TARGET nitemare3d POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different ${N3D_HUD_PAK} $<TARGET_FILE_DIR:nitemare3d>)
  endif()
else()
  message(WARNING "python3 or assets/hud missing: no hud.pak, the classic status bar is used")
endif()

target_include_directories(nitemare3d PRIVATE src third_party/nuked_opl3 third_party/tinysoundfont third_party/stb_vorbis)
if(EMSCRIPTEN)
  set(N3D_WASM_FLAGS -sUSE_SDL=2)
  if(N3D_WASM_THREADS)
    list(APPEND N3D_WASM_FLAGS -pthread)
  endif()
  target_compile_options(nitemare3d PRIVATE ${N3D_WASM_FLAGS})
  target_link_options(nitemare3d PRIVATE ${N3D_WASM_FLAGS}
    -sALLOW_MEMORY_GROWTH=1
    -sINITIAL_MEMORY=128MB
    -sSTACK_SIZE=8MB
    -sINVOKE_RUN=0
    -sEXIT_RUNTIME=0
    -sFORCE_FILESYSTEM=1
    -sMODULARIZE=1
    -sEXPORT_NAME=createNitemare3D
    "-sEXPORTED_RUNTIME_METHODS=['callMain','FS','IDBFS']"
    -lidbfs.js)
  if(N3D_WASM_THREADS)
    target_link_options(nitemare3d PRIVATE "-sPTHREAD_POOL_SIZE=Math.min(Math.max(navigator.hardwareConcurrency-1,0),11)")
  endif()
  if(N3D_WASM_DATA)
    target_link_options(nitemare3d PRIVATE --preload-file "${N3D_WASM_DATA}@/bundled")
  endif()
  if(N3D_HUD_PAK)
    target_link_options(nitemare3d PRIVATE --preload-file "${N3D_HUD_PAK}@/hud.pak")
  endif()
  set_target_properties(nitemare3d PROPERTIES SUFFIX ".js")
  # page shell next to nitemare3d.js / .wasm
  foreach(f index.html n3d-web.js n3d-web.css)
    configure_file(${CMAKE_SOURCE_DIR}/web/${f} ${CMAKE_BINARY_DIR}/${f} COPYONLY)
  endforeach()
elseif(TARGET SDL2::SDL2)
  if(TARGET SDL2::SDL2main)
    target_link_libraries(nitemare3d PRIVATE SDL2::SDL2main)
  endif()
  target_link_libraries(nitemare3d PRIVATE SDL2::SDL2)
else()
  target_include_directories(nitemare3d PRIVATE ${SDL2_INCLUDE_DIRS})
  target_link_libraries(nitemare3d PRIVATE ${SDL2_LIBRARIES})
endif()

if(N3D_D3D12)
  # d3d12, dxgi and d3dcompiler are loaded at run time; only the interface GUIDs are linked.
  target_sources(nitemare3d PRIVATE src/video_d3d12.c)
  target_compile_definitions(nitemare3d PRIVATE N3D_D3D12)
  target_link_libraries(nitemare3d PRIVATE dxguid)
endif()

if(WIN32 AND TARGET SDL2::SDL2)
  # the game runs straight from the build folder
  add_custom_command(TARGET nitemare3d POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL2::SDL2> $<TARGET_FILE_DIR:nitemare3d>)
endif()

if(NOT MSVC)
  target_link_libraries(nitemare3d PRIVATE m)
endif()
if(Threads_FOUND)
  target_link_libraries(nitemare3d PRIVATE Threads::Threads)
endif()

if(WIN32)
  set_target_properties(nitemare3d PROPERTIES WIN32_EXECUTABLE TRUE)
  target_link_libraries(nitemare3d PRIVATE ws2_32)
endif()

if(MSVC)
  target_compile_definitions(nitemare3d PRIVATE _CRT_SECURE_NO_WARNINGS strcasecmp=_stricmp)
  set_source_files_properties(${GEN_SOURCES} PROPERTIES COMPILE_OPTIONS "/w;/bigobj")
else()
  target_compile_options(nitemare3d PRIVATE -fno-strict-aliasing)
  # The generated files are huge; -O1 keeps build times sane and is plenty fast.
  set_source_files_properties(${GEN_SOURCES} PROPERTIES COMPILE_OPTIONS "-w;-O1")
endif()
if(N3D_TRACE)
  target_compile_definitions(nitemare3d PRIVATE RT_TRACE)
endif()

if(EMSCRIPTEN)
  return()
endif()

# Offline renderer: SND.DAT tunes -> music/tuneNN.wav/.ogg (see README, "Pre-rendered music").
add_executable(n3d-render-music
  tools/render_music.c
  src/music.c
  src/dls.c
  src/msadlib.c
  third_party/stb_vorbis/stb_vorbis.c
  third_party/nuked_opl3/opl3.c)
target_include_directories(n3d-render-music PRIVATE src third_party/nuked_opl3 third_party/tinysoundfont third_party/stb_vorbis)
if(NOT MSVC)
  target_link_libraries(n3d-render-music PRIVATE m)
endif()
if(MSVC)
  target_compile_definitions(nitemare3d PRIVATE strncasecmp=_strnicmp)
  target_compile_definitions(n3d-render-music PRIVATE _CRT_SECURE_NO_WARNINGS strcasecmp=_stricmp strncasecmp=_strnicmp)
  set_source_files_properties(third_party/stb_vorbis/stb_vorbis.c PROPERTIES COMPILE_OPTIONS "/w")
else()
  set_source_files_properties(third_party/stb_vorbis/stb_vorbis.c PROPERTIES COMPILE_OPTIONS "-w")
endif()
